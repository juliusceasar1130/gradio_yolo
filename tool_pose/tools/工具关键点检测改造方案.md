def get_tool_pose_config(self) -> Dict[str, Any]:
    """从tool_pose.yaml读取工具关键点配置"""
    import yaml
    from pathlib import Path
    
    tool_pose_yaml = Path(__file__).parent.parent.parent / 'datasets' / 'tools' / 'tool_pose.yaml'
    
    try:
        with open(tool_pose_yaml, 'r', encoding='utf-8') as f:

        # 解析配置并返回结构化数据

            'num_classes': config.get('nc', 2),

TOOL_POSE_CONFIG = None

def get_tool_pose_config_cached():
    """获取缓存的工具关键点配置"""
    global TOOL_POSE_CONFIG
    if TOOL_POSE_CONFIG is None:
        TOOL_POSE_CONFIG = get_pose_config()
    return TOOL_POSE_CONFIG

def get_keypoint_names_for_class(class_id: int = None):
    """根据类别ID获取关键点名称"""
    config = get_tool_pose_config_cached()
    kpt_names_dict = config.get('kpt_names', {})
    if class_id is not None and class_id in kpt_names_dict:
        return kpt_names_dict[class_id]
    return config.get('default_kpt_names', [])

#### 3.3 修改validate_pose_config()函数

**原代码**（第92-127行）：

**新代码**：
```python
def validate_tool_pose_config(config: Dict[str, Any]):
    """验证工具关键点配置的有效性"""
    try:
        kpt_shape = config.get('kpt_shape', [7, 3])
        num_kpts = kpt_shape[0]
        
        # 验证关键点形状
        if not isinstance(kpt_shape, list) or len(kpt_shape) != 2:
            raise ValueError(f"kpt_shape必须是[数量, 维度]格式")
        
        # 验证关键点名称
        kpt_names = config.get('kpt_names', {})
        for class_id, names in kpt_names.items():
            if len(names) != num_kpts:
                raise ValueError(f"类别{class_id}的关键点数量({len(names)})与kpt_shape({num_kpts})不匹配")
        
        # 验证骨架连接
        skeleton = config.get('skeleton', [])
        for connection in skeleton:
            if not isinstance(connection, list) or len(connection) != 2:
                raise ValueError(f"骨架连接格式错误: {connection}")
            if connection[0] >= num_kpts or connection[1] >= num_kpts:
                raise ValueError(f"骨架连接索引超出范围: {connection}")
        
        # 验证flip_idx
        flip_idx = config.get('flip_idx', [])
        if flip_idx and len(flip_idx) != num_kpts:
            raise ValueError(f"flip_idx长度({len(flip_idx)})与关键点数量({num_kpts})不匹配")
        
        return True
        
    except Exception as e:
        logger.error(f"工具关键点配置验证失败: {e}")
        raise
```

**需要新增的方法**：
```python
def _get_object_class_id(self, obj_idx: int) -> int:
    """从检测结果中获取对象的类别ID"""
    try:
        if hasattr(self.raw_result, 'boxes') and self.raw_result.boxes is not None:
            cls_data = self.raw_result.boxes.cls
            if cls_data is not None and len(cls_data) > obj_idx:
                return int(cls_data[obj_idx].cpu().item())
    except Exception as e:
        logger.warning(f"获取对象类别ID失败: {e}")
    return 0  # 默认返回tool1
```
